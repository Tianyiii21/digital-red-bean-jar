function loadClasses() { return JSON.parse(localStorage.getItem(CLASSES_KEY) || '[]'); }

let selectedClass = null;

function renderClassList() {
  const list = document.getElementById('class-list');
  const classes = loadClasses();
  if (!classes.length) {
    list.innerHTML = '<div class="empty-hint">No classes yet — add one above</div>';
    document.getElementById('btn-launch').disabled = true;
    return;
  }
  list.innerHTML = classes.map(c => `
    <div class="class-item ${selectedClass === c ? 'selected' : ''}" data-name="${c}">
      <div class="class-radio"></div>
      <div class="class-name">${c}</div>
      <div class="class-beans">${getBeans(c)} beans</div>
      <button class="class-delete" data-delete="${c}">×</button>
    </div>`).join('');

  list.querySelectorAll('.class-item').forEach(item => {
    item.addEventListener('click', e => {
      if (e.target.classList.contains('class-delete')) return;
      selectedClass = item.dataset.name;
      renderClassList();
      document.getElementById('btn-launch').disabled = false;
    });
  });
  list.querySelectorAll('.class-delete').forEach(btn => {
    btn.addEventListener('click', e => {
      e.stopPropagation();
      const name = btn.dataset.delete;
      if (!confirm(`Remove "${name}"? Bean count will be lost.`)) return;
      const classes = loadClasses().filter(c => c !== name);
      saveClasses(classes);
      if (selectedClass === name) { selectedClass = null; document.getElementById('btn-launch').disabled = true; }
      renderClassList();
    });
  });
}

document.getElementById('btn-add-class').addEventListener('click', () => {
  const input = document.getElementById('class-input');
  const name = input.value.trim();
  if (!name) return;
  const classes = loadClasses();
  if (!classes.includes(name)) { classes.push(name); saveClasses(classes); }
  selectedClass = name; input.value = '';
  renderClassList();
  document.getElementById('btn-launch').disabled = false;
});

  if (e.key === 'Enter') document.getElementById('btn-add-class').click();
});

// ── Bean rendering: jittered hex packing ─────────────────
// Pre-compute all bean positions once using a seeded RNG.
// Beans are placed in hex-offset rows from the bottom up,
// each nudged by a small random jitter so they look hand-poured.

function seededRand(seed) {
  let s = seed >>> 0;
  return () => {
    s = Math.imul(s ^ (s >>> 15), s | 1);
    s ^= s + Math.imul(s ^ (s >>> 7), s | 61);
    return ((s ^ (s >>> 14)) >>> 0) / 4294967296;
  };
}

// Build the full list of 300 bean positions once
function buildBeanPositions() {
  const rand = seededRand(1337);
  const positions = [];

  const BEAN_R  = 4.8;
  const BEAN_RY = 4.2;
  const COL_STEP = BEAN_R * 2 + 1.0;   // horizontal spacing
  const ROW_STEP = BEAN_RY * 2 + 0.6;  // vertical spacing
  const JITTER   = 2.2;                 // max random nudge
  const MARGIN   = 3;

  const cols = Math.floor((JAR_W - MARGIN * 2) / COL_STEP);
  const startX = JAR_X + MARGIN + (JAR_W - MARGIN * 2 - (cols * COL_STEP - 1.0)) / 2;

  let row = 0;
  while (positions.length < MAX_BEANS) {
    const offset = (row % 2 === 0) ? 0 : COL_STEP * 0.5;
    const baseY  = JAR_BOTTOM - BEAN_RY - 1 - row * ROW_STEP;
    if (baseY - BEAN_RY < JAR_TOP) break;

    for (let col = 0; col < cols && positions.length < MAX_BEANS; col++) {
      const baseX = startX + col * COL_STEP + offset;
      if (baseX + BEAN_R > JAR_X + JAR_W - MARGIN) continue;

      const jx = (rand() - 0.5) * JITTER;
      const jy = (rand() - 0.5) * JITTER;
      const sizeV = 0.88 + rand() * 0.24; // 0.88–1.12 size variation
      const grad  = Math.floor(rand() * 3);

      positions.push({
        x: baseX + jx,
        y: baseY  + jy,
        rx: BEAN_R  * sizeV,
        ry: BEAN_RY * sizeV,
        grad,
      });
    }
    row++;
  }

  return positions;
}

const BEAN_POSITIONS = buildBeanPositions();

function renderJar(doc, beanCount) {
  const bg  = doc.getElementById('bean-fill-bg');
  const grp = doc.getElementById('beans-group');
  if (!bg || !grp) return;

  const fraction = Math.min(beanCount / MAX_BEANS, 1);
  const fillH    = fraction * JAR_H;
  const fillY    = JAR_BOTTOM - fillH;

  bg.setAttribute('y', fillY - 1);
  bg.setAttribute('height', Math.max(fillH + 2, 0));

  if (beanCount === 0) { grp.innerHTML = ''; return; }

  const count = Math.min(beanCount, BEAN_POSITIONS.length);
  let svg = '';

  for (let i = 0; i < count; i++) {
    const { x, y, rx, ry, grad } = BEAN_POSITIONS[i];
    // Skip beans above current fill level
    if (y - ry < fillY - 2) continue;

    svg += `<ellipse cx="${x.toFixed(1)}" cy="${y.toFixed(1)}" rx="${rx.toFixed(1)}" ry="${ry.toFixed(1)}" fill="url(#bg${grad})"/>`;
    // Specular highlight
    svg += `<ellipse cx="${(x - rx*0.28).toFixed(1)}" cy="${(y - ry*0.30).toFixed(1)}" rx="${(rx*0.30).toFixed(1)}" ry="${(ry*0.24).toFixed(1)}" fill="rgba(255,255,255,0.19)"/>`;
  }

  grp.innerHTML = svg;
}

// ── Cartoon popup ─────────────────────────────────────────
let cartoonTimer = null;
function showCartoon(doc, svgStr) {
  const popup = doc.getElementById('cartoon-popup');
  if (!popup) return;
  clearTimeout(cartoonTimer);
  popup.innerHTML = svgStr;
  const inner = popup.firstElementChild;
  if (inner) { inner.style.animation = 'none'; void inner.offsetWidth; inner.style.animation = 'cartoonPop 0.5s cubic-bezier(0.34,1.56,0.64,1) forwards'; }
  popup.classList.add('show');
  cartoonTimer = setTimeout(() => {
    if (inner) inner.style.animation = 'cartoonFade 0.35s ease-in forwards';
    setTimeout(() => { popup.classList.remove('show'); popup.innerHTML = ''; }, 380);
  }, 2800);
}

// ── Wooden tap sound (Web Audio API, no files needed) ────
let audioCtx = null;

function playWoodenTap() {
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

    const ctx = audioCtx;
    const now = ctx.currentTime;

    // Layer 1: short noise burst — the "tap" transient
    const bufferSize = ctx.sampleRate * 0.06; // 60ms of noise
    const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
    const data   = buffer.getChannelData(0);
    for (let i = 0; i < bufferSize; i++) {
      data[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / bufferSize, 6);
    }
    const noise = ctx.createBufferSource();
    noise.buffer = buffer;

    // Bandpass: shape noise into a light tick (~1200Hz center)
    const bp = ctx.createBiquadFilter();
    bp.type = 'bandpass';
    bp.frequency.value = 1200;
    bp.Q.value = 3.5;

    // High-pass to keep it thin and bright
    const hp = ctx.createBiquadFilter();
    hp.type = 'highpass';
    hp.frequency.value = 600;

    // Gain envelope: very fast attack, very short decay
    const gain = ctx.createGain();
    gain.gain.setValueAtTime(0, now);
    gain.gain.linearRampToValueAtTime(0.30, now + 0.002);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.03);

    noise.connect(bp);
    bp.connect(hp);
    hp.connect(gain);
    gain.connect(ctx.destination);
    noise.start(now);
    noise.stop(now + 0.04);

    // Layer 2: light high tone (~700Hz) — thin click body, no low warmth
    const osc = ctx.createOscillator();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(700, now);
    osc.frequency.exponentialRampToValueAtTime(400, now + 0.03);

    const oscGain = ctx.createGain();
    oscGain.gain.setValueAtTime(0, now);
    oscGain.gain.linearRampToValueAtTime(0.10, now + 0.002);
    oscGain.gain.exponentialRampToValueAtTime(0.001, now + 0.03);

    osc.connect(oscGain);
    oscGain.connect(ctx.destination);
    osc.start(now);
    osc.stop(now + 0.04);

  } catch(e) {
    // Silently fail if audio isn't available
  }
}

// ── Wire jar ──────────────────────────────────────────────
function wireJar(doc, className) {
  let beanCount = getBeans(className);
  let shown     = getShown(className);

  doc.getElementById('pip-class-label').textContent = className;
  renderJar(doc, beanCount);

  const jarSvg = doc.getElementById('jar-svg');

  function bounce() {
    if (!jarSvg) return;
    jarSvg.style.animation = 'none'; void jarSvg.offsetWidth;
    jarSvg.style.animation = 'jarBounce 0.4s ease';
  }

  function checkMilestones(prev) {
    for (const m of MILESTONES) {
      if (prev < m.beans && beanCount >= m.beans && !shown[m.key]) {
        shown[m.key] = true; setShown(className, shown);
        showCartoon(doc, m.cartoon);
      }
      if (beanCount < m.beans && shown[m.key]) { delete shown[m.key]; setShown(className, shown); }
    }
  }

  doc.getElementById('btn-add').addEventListener('click', () => {
    const prev = beanCount;
    beanCount = Math.min(beanCount + 1, MAX_BEANS);
    setBeans(className, beanCount); renderJar(doc, beanCount);
    checkMilestones(prev); bounce(); renderClassList();
    playWoodenTap();
  });

  doc.getElementById('btn-remove').addEventListener('click', () => {
    const prev = beanCount;
    beanCount = Math.max(beanCount - 1, 0);
    setBeans(className, beanCount); renderJar(doc, beanCount);
    checkMilestones(prev); renderClassList();
  });

  const addBtn = doc.getElementById('btn-add');
  const remBtn = doc.getElementById('btn-remove');
  addBtn.addEventListener('mouseenter', () => { addBtn.style.background = '#a83020'; addBtn.style.transform = 'translateY(-1px)'; });
  addBtn.addEventListener('mouseleave', () => { addBtn.style.background = '#7a1f0e'; addBtn.style.transform = ''; });
  remBtn.addEventListener('mouseenter', () => { remBtn.style.background = 'rgba(245,220,180,0.25)'; remBtn.style.color = 'rgba(245,220,180,1)'; });
  remBtn.addEventListener('mouseleave', () => { remBtn.style.background = 'rgba(245,220,180,0.15)'; remBtn.style.color = 'rgba(245,220,180,0.75)'; });
}
